param code:string -> string
//check if any error in code
&simtalkcode.program := code
var errMsg : string
var errLine : integer
var hasError := &simtalkcode.hasSyntaxError(errMsg, errLine)
if hasError
	action_result["result"] := "failed"
    return " hasError ： "+ errMsg+ " (in row :"+errLine+")"

else
	action_result["result"] := "success"
	return  "has no Error"
end
