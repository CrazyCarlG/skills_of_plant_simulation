    def configureSequenceTables(self):
        """
        _______________________________________________________________________________

        Method Tasks:
            Ensures that each sequence table in the Input has the required “Orig” and “Chrom” 
            columns, adding them if necessary. This is needed so evolutionary algorithms can track 
            and manipulate original and chromosome-based indices consistently.
        _______________________________________________________________________________
        """
    
        for idx in range(current.Input.yDim):
            attribute = eval(current.Input[1,idx+1])
            if type(attribute) is not int:
                table = attribute
                
                table.ColumnIndex = True
                
                column = table.xDim
                  
                # WHY: “Orig” column stores the original order of the sequence.
                # This prevents losing mapping when evolutionary changes occur.
                if "Orig" not in table[1:,0][0]:
                    column = column + 1
                    table.insertColumn(column)
                    table.setDataType(column, "Integer")
                    table[column, 0] = "Orig"
                    
                    for i in range(table.yDim+1):
                        table[column,i] = i   
                
                # WHY: “Chrom” column allows genetic algorithms to map simulation data
                # back to chromosome indices for traceability.
                if "Chrom" not in table[1:,0][0]:
                    column = column + 1
                    table.insertColumn(column)
                    table.setDataType(column, "Integer")
                    table[column, 0] = "Chrom"
                    
                    for i in range(table.yDim+1):
                        table[column,i] = i

    def writeSimTalkFile(self, parameters, t_idx):
        """
        _______________________________________________________________________________

        Method Tasks:
            Generates a temporary SimTalk script (.txt) from a template, inserting all

            This script is later passed to the Plant Simulation console for execution.
        _______________________________________________________________________________
        """
        
        template = current.SimTemplate.Program.replace("---","")
    
        def _default(o):
            if isinstance(o, (np.integer, np.floating)):
                return o.item()
            if isinstance(o, np.ndarray):
                return o.tolist()
            raise TypeError(f"Type {type(o)} not serializable")
            
        a = parameters["sequences"]
    
        values = {
            "eventController": parameters["eventController"],
            "buffers":          parameters["buffers"],
            "output":           parameters["output"],
            "capacityValues":   parameters["capacityValues"],
            "dataTables":       parameters["dataTables"],
            "sequences":        parameters["sequences"],
        }
    
        ser = {
            k: (
                "[" + ", ".join(v) + "]"
                if isinstance(v, list) and all(isinstance(i, str) for i in v)
                else json.dumps(v, default=_default)
                if isinstance(v, (list, dict, np.ndarray))
                else v
                if isinstance(v, str)
                else str(v)
            )
            for k, v in values.items()
        }
    
        filled = template.format_map(ser)

        out_path = current_working_dir + rf"\simtalk{t_idx}.txt"
        with open(out_path, "w", encoding="utf-8") as f:
            f.write(filled)

    def string_to_dict(self, s: str) -> dict:
        result = {}
        for line in s.splitlines():
            line = line.strip()
            if not line:
                continue
            parts = [p.strip() for p in line.split(":", 2)]
            if len(parts) == 3:
                parent, child, value = parts
                if parent not in result or not isinstance(result[parent], dict):
                    result[parent] = {}
                result[parent][child] = value
            elif len(parts) == 2:
                key, value = parts
                result[key] = value
        return result
