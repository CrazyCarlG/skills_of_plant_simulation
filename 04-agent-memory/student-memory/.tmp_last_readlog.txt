class HTMLHandler:
    """
    _______________________________________________________________________________

    Class Tasks:
        Create SVG plots (min/mean/max per objective + Pareto) and produce HTML in 
        HTMLReport. This is done to visualize the Optimization to the user.
    _______________________________________________________________________________
            
    Used by:
        Optimize
        
    Uses:
        HTMLReport
    _______________________________________________________________________________ 
        
    """


    def __init__(self):
        self.history = {"min": [], "mean": [], "max": []}

    def create_min_max_avg_plot(self,F, generation):
        """
        Method Tasks:
            Make min/mean/max plots per objective and update history.
        """
        
        F = np.atleast_2d(F)
        n_obj = F.shape[1]

        f_min = F.min(axis=0)
        f_mean = F.mean(axis=0)
        f_max = F.max(axis=0)

        self.history["min"].append(f_min.copy())
        self.history["mean"].append(f_mean.copy())
        self.history["max"].append(f_max.copy())
    
        mins = np.array(self.history["min"])
        means = np.array(self.history["mean"])
        maxs = np.array(self.history["max"])
        generations = np.arange(1, len(mins) + 1)
    
        fig, axes = plt.subplots(1, n_obj, figsize=(6 * n_obj, 4))
        if n_obj == 1:
            axes = [axes]
            
        objectives = ["Throughput","Cost"]    
    
        for i, ax in enumerate(axes):
            y_min = mins[:, i]
            y_mean = means[:, i]
            y_max = maxs[:, i]

            y_min_plot = np.abs(y_min)
            y_mean_plot = np.abs(y_mean)
            y_max_plot = np.abs(y_max)

            min_color = "green" if np.all(y_min >= 0) else "red"
            max_color = "red" if np.all(y_max >= 0) else "green"
            mean_color = "blue"
            
            if not np.all(y_min >= 0):
                temp = y_min_plot
                y_min_plot = y_max_plot
                y_max_plot = temp
    
            ordered = [("Minimum", y_min_plot, min_color),
                       ("Mean", y_mean_plot, mean_color),
                       ("Maximum", y_max_plot, max_color)]

            for label, y, color in ordered:
                ax.plot(generations, y, label=label, color=color, linewidth=2)
    
            ax.set_xlabel("Generation")
            ax.set_ylabel(objectives[i])
            ax.legend(framealpha=0.85, loc="best")
            ax.xaxis.set_major_locator(MaxNLocator(integer=True))
    
        plt.tight_layout()
        return fig
    
    def create_pareto_plot(self,pareto_front):
        """
        Method Tasks:
            Create a Pareto front scatter plot (mirror negatives if needed).
        """
        
        pareto_front = np.array(pareto_front)
    
        mirrored = pareto_front.copy()
        if np.all(pareto_front[:, 0] <= 0):
            mirrored[:, 0] = -mirrored[:, 0]
        if np.all(pareto_front[:, 1] <= 0):
            mirrored[:, 1] = -mirrored[:, 1]
    
        fig, ax = plt.subplots(figsize=(6,4))
    
        ax.scatter(
            mirrored[:, 0],
            mirrored[:, 1],
            s=30,
            facecolors="none",
            edgecolors="blue",
            label="Current Pareto Front"
        )
    
        ax.set_title("Pareto Front")
        ax.set_xlabel("Throughput")
        ax.set_ylabel("Cost")
        ax.legend(framealpha=0.85)
        plt.tight_layout()
    
        return fig

from pymoo.core.crossover import Crossover

class GroupwiseCrossover(Crossover):
    """
    _______________________________________________________________________________

    Class Tasks:
        Provide a crossover operator that applies different crossover strategies
        to distinct groups of genes within individuals. Each gene group is
        defined by a mask value; a dedicated crossover operator from
        `group_crossovers` is used to combine only the genes that belong to
        that group. This enables heterogeneous recombination where segments
        with different semantics (e.g. integer buffers, real-valued parameters,
        permutation segments) are processed by the most appropriate crossover.
        This is needed, since buffers and sequences have to be handled differently
    _______________________________________________________________________________
    
    Used by:
        Optimizer
        
    Uses:
        -
    _______________________________________________________________________________
    """
    
    def __init__(self, mask, group_crossovers):
        n_parents_max = max(crossover.n_parents for crossover in group_crossovers.values())
        n_offsprings_max = max(crossover.n_offsprings for crossover in group_crossovers.values())
        
        super().__init__(n_parents=n_parents_max, n_offsprings=n_offsprings_max)
        
        self.mask = np.array(mask)
        self.group_crossovers = group_crossovers
        
        unique_groups = np.unique(self.mask)
        for group in unique_groups:
            if group not in group_crossovers:
                raise ValueError(f"Kein Crossover-Operator für Gruppe '{group}' angegeben")
    
    def _do(self, problem, X, **kwargs):
        """
        _______________________________________________________________________________

        Method Tasks:
            Execute group-wise crossover on the provided parents array `X`. For each
            mating (pairing) the method:
            - Creates an offspring template by copying parents.
            - Iterates over each unique group defined by `self.mask`.
            - For each group, extracts the subgroup of genes from the parents,
              constructs a small subproblem describing bounds for that subgroup,
              and delegates the recombination to the group's crossover operator.
            - Inserts the resulting subgroup offspring back into the global offspring
              template so full-length offspring are constructed using per-group logic.
            The method returns a properly shaped offspring array expected by pymoo.
        _______________________________________________________________________________
        """
        n_parents, n_matings, n_vars = X.shape
        
        Y = np.empty((self.n_offsprings, n_matings, n_vars), dtype=X.dtype)
        
        for i in range(n_matings):
            offsprings = [X[j, i].copy() for j in range(self.n_offsprings)]
            
            for group in np.unique(self.mask):
                crossover = self.group_crossovers[group]
                
                group_indices = np.where(self.mask == group)[0]
                
                if len(group_indices) == 0:
                    continue
                
                class SubProblem:
                    def __init__(self, n_var, xl=None, xu=None):
                        self.n_var = n_var
                        self.xl = xl
                        self.xu = xu
                    
                    def bounds(self):
                        return self.xl, self.xu
                        
                    def has_bounds(self):
                        return True    
                
                xl = None
                xu = None
                
                if hasattr(problem, 'bounds'):
                    problem_xl, problem_xu = problem.bounds()
                    if problem_xl is not None and hasattr(problem_xl, '__iter__'):
                        xl = np.array([problem_xl[i] for i in group_indices])
                    else:
                        xl = problem_xl
                        
                    if problem_xu is not None and hasattr(problem_xu, '__iter__'):
                        xu = np.array([problem_xu[i] for i in group_indices])
                    else:
                        xu = problem_xu
                elif hasattr(problem, 'xl') and hasattr(problem, 'xu'):
                    if problem.xl is not None and hasattr(problem.xl, '__iter__'):
                        xl = np.array([problem.xl[i] for i in group_indices])
                    else:
                        xl = problem.xl
                        
                    if problem.xu is not None and hasattr(problem.xu, '__iter__'):
                        xu = np.array([problem.xu[i] for i in group_indices])
                    else:
                        xu = problem.xu
                
                sub_problem = SubProblem(len(group_indices), xl, xu)
                
                X_sub = np.zeros((crossover.n_parents, 1, len(group_indices)), dtype=X.dtype)
                for j in range(min(crossover.n_parents, n_parents)):
                    X_sub[j, 0] = X[j, i, group_indices]
                
                Y_sub = crossover._do(sub_problem, X_sub, **kwargs)
                
                for j in range(min(crossover.n_offsprings, self.n_offsprings)):
                    offsprings[j][group_indices] = Y_sub[j, 0]
            
            for j in range(self.n_offsprings):
                Y[j, i] = offsprings[j]
        
        return Y

from pymoo.core.mutation import Mutation

class GroupwiseMutation(Mutation):
    """
    _______________________________________________________________________________

    Class Tasks:
        Implement a mutation operator that applies different mutation strategies
        to distinct groups of variables as defined by a mask. Each group is
        mapped to a dedicated mutation operator in `group_mutations`. During
        mutation, the operator extracts group-specific variable segments,
        constructs a lightweight SubProblem describing that segment (bounds and
        variable types if available), and delegates the actual mutation to the
        group's operator. The resulting subgroup outputs are merged back into
        full-length individuals.

        This enables mixed-semantics mutation (e.g. integer/permutation parts,
        continuous parts) within pymoo optimization flows and is needed, since 
        buffers and sequences have to be handled differently
    _______________________________________________________________________________

    Used by:
       Optimizer

    Uses:
        -
    _______________________________________________________________________________
    """
    
    def __init__(self, mask, group_mutations):
        super().__init__()
        
        self.mask = np.array(mask)
        self.group_mutations = group_mutations
        
        
        unique_groups = np.unique(self.mask)
        for group in unique_groups:
            if group not in group_mutations:
                raise ValueError(f"Kein Mutations-Operator für Gruppe '{group}' angegeben")
    
    def _do(self, problem, X, **kwargs):
        """
        _______________________________________________________________________________

        Method Tasks:
            Apply group-wise mutation to a 2D population array `X`. For each mask-defined
            group the method:
              - Identifies indices of variables that belong to the group.
              - Builds a minimal SubProblem describing the group's variable count and bounds.
              - Calls the group's mutation operator for each individual (or batch),
                receiving a mutated subgroup.
              - Reintegrates the mutated subgroup into the output population array.
            The method returns the mutated population with the same shape as `X`.
        _______________________________________________________________________________
        """
        
        if X.ndim != 2:
            raise ValueError(f"Erwartete Form (n_individuals, n_vars), erhielt {X.shape}")
        
        n_individuals, n_vars = X.shape
        
        Y = X.copy()

        for group in np.unique(self.mask):
            mutation = self.group_mutations[group]
            
            group_indices = np.where(self.mask == group)[0]
            
            if len(group_indices) == 0:
                continue
            
            class SubProblem:
                def __init__(self, n_var, xl=None, xu=None):
                    self.n_var = n_var
                    self.xl = xl  
                    self.xu = xu
                    self._xl = xl 
                    self._xu = xu
                
                def bounds(self):
                    return self._xl, self._xu
            
            xl = None
            xu = None
            
            if hasattr(problem, 'bounds'):
                problem_xl, problem_xu = problem.bounds()
                if problem_xl is not None and hasattr(problem_xl, '__iter__'):
                    xl = np.array([problem_xl[i] for i in group_indices])
                else:
                    xl = problem_xl
                    
                if problem_xu is not None and hasattr(problem_xu, '__iter__'):
                    xu = np.array([problem_xu[i] for i in group_indices])
                else:
                    xu = problem_xu
            elif hasattr(problem, 'xl') and hasattr(problem, 'xu'):
                if problem.xl is not None and hasattr(problem.xl, '__iter__'):
                    xl = np.array([problem.xl[i] for i in group_indices])
                else:
                    xl = problem.xl
                    
                if problem.xu is not None and hasattr(problem.xu, '__iter__'):
                    xu = np.array([problem.xu[i] for i in group_indices])
                else:
                    xu = problem.xu
            
            sub_problem = SubProblem(len(group_indices), xl, xu)
            
            for i in range(n_individuals):
                x_i = X[i, group_indices].reshape(1, -1)  
                
                try:
                    y_i = mutation._do(sub_problem, x_i, **kwargs)
                    
                    Y[i, group_indices] = y_i.flatten() 
                except Exception as e:
                    print(f"Fehler bei Mutation für Gruppe {group}: {e}")
                    continue
        
        return Y

from pymoo.core.sampling import Sampling

class GroupwiseSampling(Sampling):
    """
    _______________________________________________________________________________

    Class Tasks:
        Provide a sampling operator that applies different sampling strategies to
        distinct groups of variables as defined by a mask. For each group, the
        corresponding sampling operator in `group_samplings` is used to generate
        samples for that subset of variables. The produced subgroup samples are
        validated, reshaped if necessary, and assembled into full-length
        individuals consistent with the problem's bounds and variable types.
        This enables mixed-semantics initialization (e.g., integer permutations,
        continuous parameters) within pymoo-based optimization runs. This is 
        needed, since buffers and sequences have to be handled differently.
    _______________________________________________________________________________
    
    Used by:
        Optimizer
        
    Uses:
        -
    _______________________________________________________________________________
    """
    def __init__(self, mask, group_samplings, dtype=int):
        super().__init__()
        
        self.mask = np.array(mask)
        self.group_samplings = group_samplings
        self.dtype = dtype
        
        unique_groups = np.unique(self.mask)
        for group in unique_groups:
            if group not in group_samplings:
                raise ValueError(f"Kein Sampling-Operator für Gruppe '{group}' angegeben")
    
    def _do(self, problem, n_samples, **kwargs):
        """
        _______________________________________________________________________________

        Method Tasks:
            Generate `n_samples` individuals by delegating sampling of each mask-defined
            variable group to its dedicated sampling operator. For every group the method:
              - Extracts the group's variable indices.
              - Constructs a lightweight SubProblem describing the subgroup's
                number of variables and bounds (xl, xu).
              - Copies variable types (vtype) for the subgroup when available.
              - Calls the group's sampling operator to produce subgroup samples.
              - Validates and reshapes the subgroup output and inserts it into
                the global sample array with the configured `dtype`.
            Returns an array (n_samples, n_vars) suitable for downstream pymoo usage.
        _______________________________________________________________________________
        
        """
        
        n_vars = problem.n_var if hasattr(problem, 'n_var') else len(self.mask)
        
        X = np.zeros((n_samples, n_vars), dtype=self.dtype)
        
        for group in np.unique(self.mask):
            sampling = self.group_samplings[group]
            
            group_indices = np.where(self.mask == group)[0]
            
            if len(group_indices) == 0:
                continue
            
            class SubProblem:
                def __init__(self, n_var, xl=None, xu=None):
                    self.n_var = n_var
                    self.xl = xl  
                    self.xu = xu
                    self._xl = xl
                    self._xu = xu
                
                def bounds(self):
                    return self._xl, self._xu
            
            xl = None
            xu = None
            
            if hasattr(problem, 'bounds'):
                problem_xl, problem_xu = problem.bounds()
                if problem_xl is not None and hasattr(problem_xl, '__iter__'):
                    xl = np.array([problem_xl[i] for i in group_indices])
                else:
                    xl = problem_xl 
                    
                if problem_xu is not None and hasattr(problem_xu, '__iter__'):
                    xu = np.array([problem_xu[i] for i in group_indices])
                else:
                    xu = problem_xu  
            elif hasattr(problem, 'xl') and hasattr(problem, 'xu'):
                if problem.xl is not None and hasattr(problem.xl, '__iter__'):
                    xl = np.array([problem.xl[i] for i in group_indices])
                else:
                    xl = problem.xl 
                    
                if problem.xu is not None and hasattr(problem.xu, '__iter__'):
                    xu = np.array([problem.xu[i] for i in group_indices])
                else:
                    xu = problem.xu 
            
            sub_problem = SubProblem(len(group_indices), xl, xu)
            
            if hasattr(problem, 'vtype') and problem.vtype is not None:
                if hasattr(problem.vtype, '__iter__'):  
                    sub_problem.vtype = np.array([problem.vtype[i] for i in group_indices])
                else:
                    sub_problem.vtype = problem.vtype
            
            X_sub = sampling._do(sub_problem, n_samples, **kwargs)
            
            if X_sub.shape != (n_samples, len(group_indices)):
                try:
                    X_sub = X_sub.reshape(n_samples, len(group_indices))
                except:
                    raise ValueError(f"Sampling für Gruppe {group} ergab Form {X_sub.shape}, "
                                    f"erwartet wurde ({n_samples}, {len(group_indices)})")
            
            X[:, group_indices] = X_sub.astype(self.dtype)
        
        return X

        for v_idx in range(4, current.Solutions.xDim):

                capacityValue = current.Solutions[v_idx,i_idx]

                        chrom_idx = c_idx +1

                if sequenceCount == table.yDim:
                    table.sort(chrom_idx,"up")

                if len(mask) >= p_idx+2 and mask[p_idx+1] != "sequence":
                    input_count = input_count +1
                    sequence_count = 0
        
        for i_idx, individual in enumerate(op.X):
            for g_idx, gene in enumerate(individual):
                current.Solutions[g_idx+4,i_idx+1] = gene
